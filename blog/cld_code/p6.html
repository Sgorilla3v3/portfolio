코드 변경사항과 기술 문서의 실시간 동기화 자동화

문서화 자동화

git diff를 분석하여 변경된 코드의 로직을 문서에 자동 반영하고, 새로운 기술 부채나 TODO를 노트로 추출합니다.

실시간 동기화

파일 감시자(Watcher)가 코드 변경을 실시간으로 감지하여 관련 기술 문서와 API 스펙을 즉시 업데이트합니다.

const { exec } = require('child_process');
const chokidar = require('chokidar');

// 1. 프로젝트 코드 변경 감시 (소스코드)
chokidar.watch('./src/**/*.{js,ts}').on('change', (filePath) => {
  console.log(`💻 코드 변경 감지: ${filePath}`);

  // Git diff를 분석하여 개발 노트 업데이트 명령 실행
  const cmd = `git diff HEAD -- "${filePath}" | claude-code "이 코드 변경사항을 분석해서:
  1. 변경된 로직 요약 및 관련 개발 노트 업데이트
  2. 새로운 기술적 학습 포인트 기록
  3. TODO 리스트 자동 갱신"`;

  exec(cmd, (error, stdout) => {
    if (!error) {
      console.log(`✅ 자동 문서화 완료: ${stdout}`);
    }
  });
});

});
